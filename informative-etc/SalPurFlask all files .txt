from flask import Flask, render_template, request, redirect, url_for, flash
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime
import csv
import os

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///database.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False
app.secret_key = 'your_secret_key'
db = SQLAlchemy(app)

# Database Models
class Supplier(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    contact = db.Column(db.String(15), nullable=False)
    address = db.Column(db.String(200), nullable=False)

class Customer(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    contact = db.Column(db.String(15), nullable=False)
    address = db.Column(db.String(200), nullable=False)

class Item(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    stock = db.Column(db.Integer, nullable=False, default=0)
    reorder_level = db.Column(db.Integer, nullable=False, default=10)

class Purchase(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    supplier_id = db.Column(db.Integer, db.ForeignKey('supplier.id'), nullable=False)
    item_id = db.Column(db.Integer, db.ForeignKey('item.id'), nullable=False)
    quantity = db.Column(db.Integer, nullable=False)
    date = db.Column(db.DateTime, nullable=False, default=datetime.utcnow)
    supplier = db.relationship('Supplier', backref='purchases')
    item = db.relationship('Item', backref='purchases')

class Sale(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    customer_id = db.Column(db.Integer, db.ForeignKey('customer.id'), nullable=False)
    item_id = db.Column(db.Integer, db.ForeignKey('item.id'), nullable=False)
    quantity = db.Column(db.Integer, nullable=False)
    date = db.Column(db.DateTime, nullable=False, default=datetime.utcnow)
    customer = db.relationship('Customer', backref='sales')
    item = db.relationship('Item', backref='sales')

# Create Database
with app.app_context():
    db.create_all()

# Routes
@app.route('/')
def index():
    return render_template('index.html')

# Supplier Routes
@app.route('/supplier', methods=['GET', 'POST'])
def supplier():
    if request.method == 'POST':
        name = request.form['name'].strip()
        contact = request.form['contact'].strip()
        address = request.form['address'].strip()
        if not name or not contact or not address:
            flash('All fields are required!', 'danger')
        elif not contact.isdigit() or len(contact) < 10:
            flash('Contact must be a valid phone number!', 'danger')
        else:
            supplier = Supplier(name=name, contact=contact, address=address)
            db.session.add(supplier)
            db.session.commit()
            flash('Supplier added successfully!', 'success')
            return redirect(url_for('supplier'))
    suppliers = Supplier.query.all()
    return render_template('supplier.html', suppliers=suppliers)

@app.route('/export_suppliers')
def export_suppliers():
    suppliers = Supplier.query.all()
    with open('static/suppliers.csv', 'w', newline='') as file:
        writer = csv.writer(file)
        writer.writerow(['ID', 'Name', 'Contact', 'Address'])
        for s in suppliers:
            writer.writerow([s.id, s.name, s.contact, s.address])
    return redirect(url_for('static', filename='suppliers.csv'))

# Item Routes
@app.route('/item', methods=['GET', 'POST'])
def item():
    if request.method == 'POST':
        name = request.form['name'].strip()
        stock = request.form['stock'].strip()
        reorder_level = request.form['reorder_level'].strip()
        if not name or not stock or not reorder_level:
            flash('All fields are required!', 'danger')
        elif not stock.isdigit() or not reorder_level.isdigit():
            flash('Stock and Reorder Level must be numbers!', 'danger')
        else:
            item = Item(name=name, stock=int(stock), reorder_level=int(reorder_level))
            db.session.add(item)
            db.session.commit()
            flash('Item added successfully!', 'success')
            return redirect(url_for('item'))
    items = Item.query.all()
    return render_template('item.html', items=items)

# Purchase Routes
@app.route('/purchase', methods=['GET', 'POST'])
def purchase():
    if request.method == 'POST':
        supplier_id = request.form['supplier_id']
        item_id = request.form['item_id']
        quantity = request.form['quantity'].strip()
        date = request.form['date']
        if not supplier_id or not item_id or not quantity or not date:
            flash('All fields are required!', 'danger')
        elif not quantity.isdigit() or int(quantity) <= 0:
            flash('Quantity must be a positive number!', 'danger')
        else:
            item = Item.query.get(item_id)
            item.stock += int(quantity)
            purchase = Purchase(supplier_id=supplier_id, item_id=item_id, quantity=int(quantity), date=datetime.strptime(date, '%Y-%m-%d'))
            db.session.add(purchase)
            db.session.commit()
            flash('Purchase recorded successfully!', 'success')
            return redirect(url_for('purchase'))
    suppliers = Supplier.query.all()
    items = Item.query.all()
    purchases = Purchase.query.all()
    return render_template('purchase.html', suppliers=suppliers, items=items, purchases=purchases)

# Sale Routes
@app.route('/sale', methods=['GET', 'POST'])
def sale():
    if request.method == 'POST':
        customer_id = request.form['customer_id']
        item_id = request.form['item_id']
        quantity = request.form['quantity'].strip()
        date = request.form['date']
        if not customer_id or not item_id or not quantity or not date:
            flash('All fields are required!', 'danger')
        elif not quantity.isdigit() or int(quantity) <= 0:
            flash('Quantity must be a positive number!', 'danger')
        else:
            item = Item.query.get(item_id)
            if item.stock < int(quantity):
                flash('Insufficient stock!', 'danger')
            else:
                item.stock -= int(quantity)
                sale = Sale(customer_id=customer_id, item_id=item_id, quantity=int(quantity), date=datetime.strptime(date, '%Y-%m-%d'))
                db.session.add(sale)
                db.session.commit()
                flash('Sale recorded successfully!', 'success')
                return redirect(url_for('sale'))
    customers = Customer.query.all()
    items = Item.query.all()
    sales = Sale.query.all()
    return render_template('sale.html', customers=customers, items=items, sales=sales)

# Reports Routes
@app.route('/reports', methods=['GET', 'POST'])
def reports():
    purchase_report = sale_report = reorder_report = []
    if request.method == 'POST':
        start_date = request.form['start_date']
        end_date = request.form['end_date']
        if not start_date or not end_date:
            flash('Both dates are required!', 'danger')
        else:
            start_date = datetime.strptime(start_date, '%Y-%m-%d')
            end_date = datetime.strptime(end_date, '%Y-%m-%d')
            # Purchase Report
            purchase_report = Purchase.query.filter(Purchase.date.between(start_date, end_date)).all()
            # Sale Report
            sale_report = Sale.query.filter(Sale.date.between(start_date, end_date)).all()
            # Item Reorder Report
            reorder_report = Item.query.filter(Item.stock <= Item.reorder_level).all()
    return render_template('reports.html', purchase_report=purchase_report, sale_report=sale_report, reorder_report=reorder_report)

if __name__ == '__main__':
    app.run(debug=True)
